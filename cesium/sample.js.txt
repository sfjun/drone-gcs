window.parseResponse = function (data) {

// ✨ [sample.js 최종 마스터 - 2D/3D 제약 해제 및 무조건 우회형 우클릭 명령창 엔진]

vw.ws3dInitCallBack = function() {
    window.viewer = ws3d.viewer;
    const scene = ws3d.viewer.scene;
    const handler = new Cesium.ScreenSpaceEventHandler(scene.canvas);

    let lastHoveredFeature = undefined;

    // 🔄 [A. 2D/3D 무관 마우스 호버 실시간 노란색 하이라이트 감시 스레드]
    handler.setInputAction(function (moveEvent) {
        if (!moveEvent.endPosition) return;
        
        // 마우스 무브 설계 모드 및 드래그 가드 해제 처리
        const pickedFeature = scene.pick(moveEvent.endPosition);

        // 🎯 3D 입체 빌딩 객체가 포인터 끝에 잡히는 순간 소방방재 체크 여부와 상관없이 무조건 즉시 하이라이트 트리거
        if (Cesium.defined(pickedFeature) && pickedFeature.primitive && !(pickedFeature.primitive instanceof Cesium.Globe)) {
            document.body.style.cursor = "pointer";
            
            if (pickedFeature !== lastHoveredFeature) {
                if (lastHoveredFeature) {
                    try { lastHoveredFeature.color = Cesium.Color.WHITE; } catch(e) {}
                }
                lastHoveredFeature = pickedFeature;
                try {
                    // 옥상 표면 노란색 마스킹 플래그 실시간 가동
                    pickedFeature.color = Cesium.Color.YELLOW.withAlpha(0.40); 
                } catch(e) {}
            }
        } else {
            // 마우스가 건물을 벗어나면 즉시 하이라이트 탈색 원복
            if (lastHoveredFeature) {
                try { lastHoveredFeature.color = Cesium.Color.WHITE; } catch (e) {}
                lastHoveredFeature = undefined;
            }
            document.body.style.cursor = "default";
        }
    }, Cesium.ScreenSpaceEventType.MOUSE_MOVE);

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // 🚨 [sample.js 최종 보정 - NaN 유령 좌표 완치 및 3중 안전 피킹 엔진]
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
        // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // 🚨 [sample.js 최종 완치 - 브라우저 우클릭 차단 및 3중 정밀 좌표 바인딩]
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    
    // 🎯 [1단계: 크롬 브라우저 자체의 흰색 기본 메뉴창 난입을 물리적으로 100% 원천 봉쇄합니다]
    scene.canvas.addEventListener("contextmenu", function (event) {
        event.preventDefault(); // 브라우저 고유 우클릭 창 취소
        event.stopPropagation(); // 이벤트 전파 가드링
        return false;
    }, { passive: false });

    handler.setInputAction(function (rightClickEvent) {
        if (!rightClickEvent.position) return;
        
        const contextMousePosition = rightClickEvent.position;
        const globe = scene.globe;
        const camera = scene.camera;
        
        // 🎯 [2단계: NaN 완치용 3중 방어선 피킹 매트릭스]
        let pickedCartesian = scene.pickPosition(contextMousePosition);
        
        if (!Cesium.defined(pickedCartesian)) {
            const pickRay = camera.getPickRay(contextMousePosition);
            pickedCartesian = globe.pick(pickRay, scene);
        }
        
        if (!Cesium.defined(pickedCartesian)) {
            pickedCartesian = camera.pickEllipsoid(contextMousePosition, globe.ellipsoid);
        }
        
        if (Cesium.defined(pickedCartesian)) {
            const cartographic = Cesium.Cartographic.fromCartesian(pickedCartesian);
            var rawLon = Cesium.Math.toDegrees(cartographic.longitude);
            var rawLat = Cesium.Math.toDegrees(cartographic.latitude);
            
            // 🎯 [3단계: 완벽한 숫자가 확보되었을 때만 GCS 관제 무선 연동 통로를 개방합니다]
            if (!isNaN(rawLon) && !isNaN(rawLat) && rawLon !== 0 && rawLat !== 0) {
                
                // 🚨 simul.js 최종 연동 규격 변수 명칭에 100% 일치하도록 전역 박제 처리!!
                window.gcsLastRightClickLon = rawLon;
                window.gcsLastRightClickLat = rawLat;
                
                console.log("🎯 [GCS 좌표 연산] 정밀 위경도 매핑 성공 ->", rawLon, rawLat);
                
                // 브라우저 흰색 창 간섭 없이 우리 메뉴창만 화면 정면에 단독 안착
                const ctxMenu = document.getElementById("gcs_context_menu");
                if (ctxMenu) {
                    ctxMenu.style.left = contextMousePosition.x + "px";
                    ctxMenu.style.top = contextMousePosition.y + "px";
                    ctxMenu.style.display = "block"; 
                    console.log("📡 [GCS 래스터 가드] 드론 관제 메뉴창 단독 앵커 개방 완료");
                }
            } else {
                console.warn("⚠️ [GCS 좌표 가드] 깨진 유령 패킷 감지로 연산을 스킵했습니다.");
            }
        }
    }, Cesium.ScreenSpaceEventType.RIGHT_CLICK);

    // 🎯 [4단계: 화면 아무 곳이나 마우스 왼쪽 클릭을 하면 관제 메뉴창이 자동으로 닫히는 가드 추가]
    scene.canvas.addEventListener("mousedown", function (e) {
        var ctxMenu = document.getElementById("gcs_context_menu");
        if (ctxMenu && e.button === 0) { // 마우스 좌클릭(0) 시
            ctxMenu.style.display = "none";
        }
    });


    // 내부 클리어 동기화 가드
    if (typeof window.initDisasterWMSLayer === "function") { window.initDisasterWMSLayer(); }
};

// 🚀 [서버 ERROR 완벽 해결본] 공간정보 데이터 서비스 API 연동 본체 함수

var wfsEvent = function (windowPosition, ecefPosition, cartographic, featureInfo, event) {
    let lon = cartographic.longitudeDD;
    let lat = cartographic.latitudeDD;
    let wktPoint = "POINT(" + lon + " " + lat + ")";

    $.ajax({
        type: "get",
        url: "https://api.vworld.kr/req/data",
        data: {
            key: VWORLD_KEY,
            // ◀ [정밀 교정 1] 로컬 테스트 도메인 가드를 풀기 위해 실제 접속 주소를 문자열로 강제 고정합니다.
            domain: "http://127.0.0.1:5500",
            service: "data",
            version: "2.0",
            request: "GetFeature",
            data: "LP_PA_CBND_BUBUN", // 연속지적도 레이어 고정 명칭
            geomFilter: wktPoint,
            geometry: "false",
            crs: "EPSG:4326"
        },
        dataType: 'jsonp',
        jsonpCallback: "parseDataResponse",
        success: function (data) {
            // 데이터 수신 결과는 전역 window.parseDataResponse 에서 통합 처리됩니다.
        },
        error: function (xhr, stat, err) {
            console.error("❌ 데이터 서비스 통신 최종 에러 발생 유형:", stat);
            console.error("상세 에러 내용:", err);
        }
    });
};

// 🚀 [서버 진짜 거부 사유 추적 전용] 데이터 API 2.0 콜백 함수
// =========================================================================
window.parseDataResponse = function (data) {
    // 1. 서버 응답 객체 전체를 마우스로 펼쳐볼 수 있게 콘솔에 강제 인쇄합니다.
    console.log("🔍 [디버깅] 브이월드 서버 응답 원본 객체 전체보기:", data);

    if (data && data.response && data.response.status === "OK") {
        if (data.response.result && data.response.result.featureCollection) {
            var features = data.response.result.featureCollection.features;
            if (features && features.length > 0) {
                var properties = features[0].properties; // 인덱스 배열 가드 추가

                let resultHtml = "";
                resultHtml += "<li> - pnu :   " + (properties.pnu || "-") + "</li>";
                resultHtml += "<li> - jibun : " + (properties.jibun || "-") + "</li>";
                resultHtml += "<li> - addr :  " + (properties.addr || "-") + "</li>";
                resultHtml += "<li> - jiga :  " + (properties.jiga || "-") + "</li>";

                $('#resultDiv').html(resultHtml);
                return;
            }
        }
        $('#resultDiv').html("<li>선택한 지점에 필지 속성 데이터가 없습니다.</li>");
    } else {
        // 2. ◀ [핵심 추적] 서버가 OK를 안 주고 ERROR를 뱉었을 때, 내부에 숨겨진 진짜 텍스트 이유를 추출합니다.
        let errorText = "알 수 없는 에러";
        let errorCode = "없음";

        if (data && data.response && data.response.error) {
            errorText = data.response.error.text || errorText;
            errorCode = data.response.error.code || errorCode;
        }

        console.error("❌ [브이월드 서버가 반환한 실제 거부 사유] 코드:", errorCode, " / 내용:", errorText);
        $('#resultDiv').html("<li style='color:red;'>서버 인증 거부: " + errorText + "</li>");
    }
};

// 🚨 [sample.js 정밀 교정 - 연속지적도 500 에러 및 타일 유실 원천 차단본]

window.vworldJijeokTileInstance = null; 

function wfsSample() {
    // 1. 🛑 중복 바인딩 및 화면 잔상 누적 방지 선제 청소
    if (window.vworldJijeokTileInstance && map) {
        try { map.getLayers().remove(window.vworldJijeokTileInstance); } catch(e) {}
    }
    if (window.viewer && window.vworldJijeokTileInstance) {
        try { window.viewer.imageryLayers.remove(window.vworldJijeokTileInstance); } catch(e) {}
    }

    // 2. 🎯 이중 인코딩 및 세슘 내장 XHR 가드를 물리적으로 우회하는 래스터 타일 프로바이더 단독 설계
    var jijeokProvider = new Cesium.WebMapServiceImageryProvider({
        url: "https://vworld.kr",
        layers: "lp_pa_cbnd_bubun,lp_pa_cbnd_bonbun", // 성공 명세서 표준 소문자 지정
        tilingScheme: new Cesium.GeographicTilingScheme(),
        parameters: {
            key: VWORLD_KEY,
            domain: "127.0.0.1",
            service: "WMS",
            version: "1.3.0",
            request: "GetMap",
            styles: "lp_pa_cbnd_bubun_webgl,lp_pa_cbnd_bonbun_webgl",
            crs: "EPSG:4326",
            format: "image/png",
            transparent: "true"
        }
    });

    // 3. 🚨 [500 에러 및 캔버스 오염 파괴 인터셉터]
    // 세슘이 주소를 조립할 때 발생하는 문자열 깨짐을 차단하고, 브이월드 전용 우회 게이트웨이를 강제 수납합니다.
    jijeokProvider.requestImage = function (x, y, level, request) {
        const rect = this.tilingScheme.tileXYToNativeRectangle(x, y, level);
        
        const baseUrl = "https://api.vworld.kr/req/wms";
        const queryParams = [
            `key=${VWORLD_KEY}`,
            `domain=127.0.0.1`,
            `service=WMS`,
            `version=1.3.0`,
            `request=GetMap`,
            `styles=lp_pa_cbnd_bubun_webgl,lp_pa_cbnd_bonbun_webgl`,
            `crs=EPSG:4326`,
            `format=image/png`,
            `transparent=true`,
            `layers=lp_pa_cbnd_bubun,lp_pa_cbnd_bonbun`,
            `bbox=${rect.west},${rect.south},${rect.east},${rect.north}`,
            `width=256`,
            `height=256`
        ];
        
        const rawTileUrl = baseUrl + "?" + queryParams.join("&");
        // 국가 공간정보 전용 프록시 관로 포워딩 기술 적용
        const proxyGateUrl = "https://map.vworld.kr/proxy.do?url=" + encodeURIComponent(rawTileUrl);

        return new Promise((resolve) => {
            const img = new Image();
            img.crossOrigin = "anonymous"; // 프록시 서버 통과용 인증 명시
            
            img.onload = function () { resolve(img); };
            img.onerror = function () {
                const empty = document.createElement("canvas");
                empty.width = 256; empty.height = 256;
                resolve(empty);
            };
            img.src = proxyGateUrl;
        });
    };

    jijeokProvider._tileDiscardPolicy = undefined;

    // 4. 2D 맵 마스터 풀과 3D 세슘 뷰어 양측 레이어 파이프에 인스턴스를 동시 안착시킵니다.
    if (window.viewer && window.viewer.imageryLayers) {
        window.vworldJijeokTileInstance = window.viewer.imageryLayers.addImageryProvider(jijeokProvider);
    } else if (map) {
        var wmsLayer = new vw.Layers();
        window.vworldJijeokTileInstance = new vw.layer.Tile(jijeokProvider);
        wmsLayer.add(window.vworldJijeokTileInstance);
    }

    chk += 1;
    // 기존 마우스 좌표 피킹(WFS) 클릭 이벤트 연동 개시
    if (map && map.onClick && typeof wfsEvent === "function") {
        map.onClick.addEventListener(wfsEvent);
    }
}

window.checkLayer = function (checkboxElement, layerCode) {
    if (!checkboxElement || !ws3d || !ws3d.viewer) return;
    
    var isVisible = checkboxElement.checked;
    var layers = ws3d.viewer.imageryLayers;
    var targetCode = String(layerCode).toLowerCase();
    
    console.log("🔍 [레이어 스캔] 타겟 코드 추적 개시 ->", targetCode);

    // 세슘 레이어 마스터 풀을 돌며 정식 영문 레이어 코드와 매칭되는 ImageryLayer를 제어합니다.
    for (var i = 0; i < layers.length; i++) {
        var layer = layers.get(i);
        if (layer && layer.imageryProvider && layer.imageryProvider._layers) {
            var currentLayerStr = String(layer.imageryProvider._layers).toLowerCase();
            
            if (currentLayerStr.includes(targetCode)) {
                layer.show = isVisible; // 레이어 실시간 물리 가시성 On/Off 제어 완수
                console.log("🎯 [GCS 인프라 완료] 레이어 풀 가시성 동기화 성공:", layerCode, "->", isVisible ? "ON" : "OFF");
            }
        }
    }
};

// 🚀 [서울시청 강제 이동 오류 완벽 해결] 현재 화면 중심 기반 2D ↔ 3D 가각 토글러
// =========================================================================
window.switchGcsMode = function (mode) {
    const btn2d = document.getElementById("btn_2d");
    const btn3d = document.getElementById("btn_3d");

    if (!ws3d || !ws3d.viewer) return;

    // 🎯 [핵심 교정] 현재 화면의 정중앙 지점 지형 좌표(ECEF)를 실시간 추출합니다.
    const canvas = ws3d.viewer.scene.canvas;
    const centerPosition = new Cesium.Cartesian2(canvas.clientWidth / 2, canvas.clientHeight / 2);
    const ray = ws3d.viewer.camera.getPickRay(centerPosition);
    const targetCartesian = ws3d.viewer.scene.globe.pick(ray, ws3d.viewer.scene);

    let centerLon = 126.9780; // 카메라 추적 실패 시 안전 가드용 기본값 (서울시청)
    let centerLat = 37.5665;
    let currentHeight = ws3d.viewer.camera.positionCartographic.height;

    if (Cesium.defined(targetCartesian)) {
        const cartographic = Cesium.Cartographic.fromCartesian(targetCartesian);
        centerLon = Cesium.Math.toDegrees(cartographic.longitude);
        centerLat = Cesium.Math.toDegrees(cartographic.latitude);
    } else {
        // 지형 픽킹에 실패할 경우 카메라 바로 밑 수직 좌표를 대안으로 추출합니다.
        const camCarto = ws3d.viewer.camera.positionCartographic;
        centerLon = Cesium.Math.toDegrees(camCarto.longitude);
        centerLat = Cesium.Math.toDegrees(camCarto.latitude);
    }

    ws3d.viewer.camera.lookAtTransform(Cesium.Matrix4.IDENTITY);

    if (mode === '2D') {
        window.isDrawingMode = true;
        if (btn2d) btn2d.className = "btn-mode active";
        if (btn3d) btn3d.className = "btn-mode";

        // 서울시청으로 안 가고, 사용자가 보던 그 화면 중심(centerLon, centerLat)으로 수직(-90도) 하강합니다.
        ws3d.viewer.camera.flyTo({
            destination: Cesium.Cartesian3.fromDegrees(centerLon, centerLat, currentHeight > 0 ? currentHeight : 350.0),
            orientation: { heading: Cesium.Math.toRadians(0.0), pitch: Cesium.Math.toRadians(-90.0), roll: 0.0 },
            duration: 1.0
        });
    } else {
        window.isDrawingMode = false;
        if (btn2d) btn2d.className = "btn-mode";
        if (btn3d) btn3d.className = "btn-mode active";

        // 사용자가 보던 그 화면 중심으로 비스듬한 조감 시야각(-45도)을 연출합니다.
        ws3d.viewer.camera.flyTo({
            destination: Cesium.Cartesian3.fromDegrees(centerLon, centerLat - 0.002, currentHeight > 0 ? currentHeight : 300.0),
            orientation: { heading: Cesium.Math.toRadians(0.0), pitch: Cesium.Math.toRadians(-45.0), roll: 0.0 },
            duration: 1.0
        });
    }
};

// 🌐 [CORS 완벽 우회 및 정형화] 실시간 법적 공역(비행제한/금지구역) 판단 함수
function checkAirspaceSafety(lon, lat) {
    const wktPoint = `POINT(${lon} ${lat})`;
    const targetLayers = "LT_C_AISPRHC,LT_C_AISRESC";

    return new Promise((resolve) => {
        $.ajax({
            type: "get",
            url: "https://api.vworld.kr/req/wms",
            data: {
                key: VWORLD_KEY,
                domain: window.location.origin,
                service: "data",
                version: "2.0",
                request: "GetFeature",
                data: targetLayers,
                geomFilter: wktPoint,
                geometry: "false",
                crs: "EPSG:4326"
            },
            dataType: "jsonp",
            jsonpCallback: "parseAirspaceResponse",
            success: function (data) {
                // 수신 레코드가 1건이라도 확인되면 해당 구역은 법적 규제 구역으로 판정하여 생성을 금지(false)합니다.
                if (data.response && data.response.status === "OK" && data.response.record && data.response.record.total > 0) {
                    resolve(false);
                } else {
                    resolve(true);
                }
            },
            error: function (xhr, stat, err) {
                resolve(true); // 통신 네트워크 순간 지연 시 작동 스크립트가 뻗지 않도록 허용값 리턴 가드 처리
            }
        });
    });
}

// 🚀 [세슘 문법 정밀 수정 완수] 드론 마커 및 수직 유도 점선 기둥선 실시간 렌더링 드로우 함수
function createDroneEntity(id, lon, lat, alt) {
    ws3d.viewer.entities.add({
        id: id,
        position: Cesium.Cartesian3.fromDegrees(lon, lat, alt),
        // fromCssString 문법 오타를 세슘 엔진 전용 규격 명칭인 fromCssColorString 으로 정상화 완료
        point: { pixelSize: 12, color: Cesium.Color.fromCssColorString("#00ffcc"), outlineColor: Cesium.Color.BLACK, outlineWidth: 2 },
        label: {
            text: `${id} (${Math.round(alt)}m)`,
            font: '12px sans-serif', verticalOrigin: Cesium.VerticalOrigin.BOTTOM, pixelOffset: new Cesium.Cartesian2(0, -12),
            fillColor: Cesium.Color.WHITE, style: Cesium.LabelStyle.FILL_AND_OUTLINE, outlineColor: Cesium.Color.BLACK, outlineWidth: 2
        },
        polyline: {
            positions: [Cesium.Cartesian3.fromDegrees(lon, lat, 0), Cesium.Cartesian3.fromDegrees(lon, lat, alt)],
            width: 2, material: new Cesium.PolylineDashMaterialProperty({ color: Cesium.Color.YELLOW })
        }
    });
}

// 10. 우측 대시보드 명단 동적 UI 인쇄 출력 스크립트
function renderWaypointListUI() {
    const container = document.getElementById("wp_list_container");
    if (!container) return;

    if (waypointList.length === 0) {
        container.innerHTML = `<span style="color:#7f8c8d; display:block; text-align:center; padding-top:40px; font-size:12px;">지도를 클릭하거나<br>JSON 파일을 불러오세요.</span>`;
        return;
    }
    container.innerHTML = "";

    waypointList.forEach((wp) => {
        const itemHtml = `
            <div class="wp-item">
                <div>
                    <strong style="color:#3498db;">${wp.id}</strong><br>
                    <span style="font-size:10px; color:#bdc3c7;">X: ${wp.lon.toFixed(4)}, Y: ${wp.lat.toFixed(4)}</span>
                </div>
                <div>
                    <input type="number" value="${Math.round(wp.alt)}" onchange="updateHeightFromInput('${wp.id}', this.value)"> m
                    <button class="btn-del" onclick="deleteWaypoint('${wp.id}')" style="margin-left:5px;">X</button>
                </div>
            </div>
        `;
        container.insertAdjacentHTML('beforeend', itemHtml);
    });
}

// 인풋 데이터 수치 박스 직접 변경 시 3D 공간 상의 마커 위치 실시간 강제 동기화 이동 함수
window.updateHeightFromInput = function (id, newHeight) {
    let wp = waypointList.find(w => w.id === id);
    if (wp) {
        wp.alt = parseFloat(newHeight);
        let entity = ws3d.viewer.entities.getById(id);
        if (entity) {
            entity.position = Cesium.Cartesian3.fromDegrees(wp.lon, wp.lat, wp.alt);
            entity.label.text = `${id} (${Math.round(wp.alt)}m)`;
            entity.polyline.positions = [Cesium.Cartesian3.fromDegrees(wp.lon, wp.lat, 0), Cesium.Cartesian3.fromDegrees(wp.lon, wp.lat, wp.alt)];
        }
    }
};

// 개별 임무 노드 삭제 버튼 클릭 시 데이터 완전 소멸 및 명단 자동 리인덱싱 재정렬 함수
window.deleteWaypoint = function (id) {
    ws3d.viewer.entities.removeById(id);
    waypointList = waypointList.filter(w => w.id !== id);

    const backupWps = [...waypointList];
    ws3d.viewer.entities.removeAll();
    waypointList = [];

    backupWps.forEach((wp, idx) => {
        const newId = `WP_${idx + 1}`;
        waypointList.push({ id: newId, lon: wp.lon, lat: wp.lat, alt: wp.alt });
        createDroneEntity(newId, wp.lon, wp.lat, wp.alt);
    });

    renderWaypointListUI();
};

// 12. [미션 I/O 관리] 최종 완성본 임무 데이터 JSON 내보내기 익스포트 함수
window.exportMissionJSON = function () {
    if (waypointList.length === 0) { alert("내보낼 임무 경로 데이터가 존재하지 않습니다."); return; }
    const fileData = waypointList.map((wp, i) => ({
        index: i + 1, id: wp.id,
        longitude_X: parseFloat(wp.lon.toFixed(6)),
        latitude_Y: parseFloat(wp.lat.toFixed(6)),
        altitude_Z: parseFloat(wp.alt.toFixed(1))
    }));
    const jsonStr = JSON.stringify(fileData, null, 2);
    const blob = new Blob([jsonStr], { type: "application/json" });
    const a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download = "vworld_gcs_mission.json";
    a.click();
    URL.revokeObjectURL(a.href);
};

// 12. [미션 I/O 관리] JSON 파일 읽기 복원 및 성공 즉시 서울시청 저고도(250m) 근접 비행 연동 함수
window.importMissionJSON = function (event) {
    const file = event.target.files;
    if (!file || !file[0]) return; // 파일 객체 단일 인덱스 타겟팅 구조 안전화
    const reader = new FileReader();
    reader.onload = function (e) {
        try {
            const parsed = JSON.parse(e.target.result);
            ws3d.viewer.entities.removeAll();
            waypointList = [];

            parsed.forEach(wp => {
                const id = wp.id || `WP_${wp.index}`;
                const lon = wp.longitude_X;
                const lat = wp.latitude_Y;
                const alt = wp.altitude_Z;
                waypointList.push({ id: id, lon: lon, lat: lat, alt: alt });
                createDroneEntity(id, lon, lat, alt);
            });
            renderWaypointListUI();
            alert(`성공적으로 파일로부터 ${waypointList.length}개의 비행 노드를 복원했습니다.`);

            // 🎯 파일을 성공적으로 가공하여 노드를 복원한 즉시 카메라를 서울시청 상공 근접(고도 250m, 가각 60도) 뷰로 가동합니다.
            ws3d.viewer.camera.flyTo({
                destination: Cesium.Cartesian3.fromDegrees(126.9780, 37.5665, 250.0),
                orientation: { heading: Cesium.Math.toRadians(0.0), pitch: Cesium.Math.toRadians(-60.0), roll: 0.0 },
                duration: 2.0
            });

        } catch (err) {
            alert("올바른 형태의 GCS 임무 JSON 파일이 아닙니다.");
        }
    };
    reader.readAsText(file[0]);
};

// [중복 선언 해결] 전역 상태 스위치 단일 정의
window.gcsPanelState = {
    isPinned: false,      // 고정핀 잠금 상태
    isCollapsed: false,   // 창 접힘 상태
    isDragging: false,    // 마우스 드래그 상태
    startX: 0, startY: 0
};

// [중복 껍데기 통합] DOMContentLoaded 이벤트 리스너를 하나로 합쳐 충돌을 차단합니다.
document.addEventListener("DOMContentLoaded", function () {
    // HTML 구조 매핑 (.gcs-sidebar 우선 타겟팅)
    const panel = document.querySelector(".gcs-sidebar") || document.querySelector(".gcs-panel") || document.getElementById("gcs_panel_container");
    if (!panel) {
        console.warn("대시보드 요소(.gcs-sidebar 또는 .gcs-panel)를 찾을 수 없습니다.");
        return;
    }

    // 드래그 조작을 할 타이틀 바 영역 지정
    let header = panel.querySelector(".gcs-header") || panel.querySelector("#gcs_panel_header");
    if (!header) {
        // 내부에 헤더 클래스가 없을 경우 최상단 타이틀(h3)을 헤더 드래그 존으로 자동 활용합니다.
        header = panel.querySelector("h3") || panel.firstElementChild;
    }

    // 📌 [기능 1] 제어 버튼 동적 생성 및 헤더 주입
    header.style.position = "relative";
    header.style.cursor = "move";

    const controlDiv = document.createElement("div");
    controlDiv.style.cssText = "position:absolute; right:10px; top:50%; transform:translateY(-50%); display:flex; gap:8px; z-index:9999;";
    controlDiv.innerHTML = `
        <button id="btn_gcs_pin" title="위치 고정/해제" style="background:none; border:none; color:#fff; cursor:pointer; font-size:14px; padding:2px; outline:none;">📌</button>
        <button id="btn_gcs_toggle" title="창 접기/펼치기" style="background:none; border:none; color:#fff; cursor:pointer; font-size:14px; padding:2px; outline:none;">🔼</button>
    `;
    header.appendChild(controlDiv);

    const pinBtn = document.getElementById("btn_gcs_pin");
    const toggleBtn = document.getElementById("btn_gcs_toggle");

    // 📌 고정핀 토글 버튼 리스너
    pinBtn.addEventListener("click", function (e) {
        e.stopPropagation(); // 헤더 드래그 이벤트 전파 방지
        window.gcsPanelState.isPinned = !window.gcsPanelState.isPinned;
        if (window.gcsPanelState.isPinned) {
            pinBtn.style.color = "#ff3838";
            pinBtn.innerText = "📍";
            header.style.cursor = "default";
        } else {
            pinBtn.style.color = "#ffffff";
            pinBtn.innerText = "📌";
            header.style.cursor = "move";
        }
    });

    // 🔼 창 접기/펼치기 토글 버튼 리스너
    const children = Array.from(panel.children);
    toggleBtn.addEventListener("click", function (e) {
        e.stopPropagation();
        window.gcsPanelState.isCollapsed = !window.gcsPanelState.isCollapsed;

        // 헤더를 제외한 모든 본문 엘리먼트들을 숨기거나 노출합니다.
        children.forEach(child => {
            if (child !== header) {
                child.style.display = window.gcsPanelState.isCollapsed ? "none" : "";
            }
        });

        if (window.gcsPanelState.isCollapsed) {
            panel.style.height = "auto";
            panel.style.minHeight = "0px";
            toggleBtn.innerText = "🔽";
        } else {
            panel.style.height = ""; // 구형 CSS 스타일 규격으로 원복
            panel.style.minHeight = "";
            toggleBtn.innerText = "🔼";
        }

        // 레이아웃 유동에 맞춰 3D 화면 왜곡을 깨끗하게 새로고침합니다.
        triggerCesiumResize();
    });

    // 🔀 마우스 드래그 앤 드롭 위치 이동 로직
    header.addEventListener("mousedown", function (e) {
        if (window.gcsPanelState.isPinned || e.target === pinBtn || e.target === toggleBtn) return;

        window.gcsPanelState.isDragging = true;
        const rect = panel.getBoundingClientRect();
        window.gcsPanelState.startX = e.clientX - rect.left;
        window.gcsPanelState.startY = e.clientY - rect.top;

        // 패널을 움직이는 동안 지도가 혼자 회전하며 튀는 현상을 일시 가드링합니다.
        if (window.viewer && window.viewer.scene) {
            window.viewer.scene.screenSpaceCameraController.enableInputs = false;
        }
    });

    document.addEventListener("mousemove", function (e) {
        if (!window.gcsPanelState.isDragging) return;

        let newLeft = e.clientX - window.gcsPanelState.startX;
        let newTop = e.clientY - window.gcsPanelState.startY;

        // 브라우저 뷰포트 밖으로 대시보드가 탈출하지 못하도록 바운딩 제한
        if (newLeft < 0) newLeft = 0;
        if (newTop < 0) newTop = 0;
        if (newLeft > window.innerWidth - panel.offsetWidth) newLeft = window.innerWidth - panel.offsetWidth;
        if (newTop > window.innerHeight - panel.offsetHeight) newTop = window.innerHeight - panel.offsetHeight;

        panel.style.position = "fixed";
        panel.style.left = newLeft + "px";
        panel.style.top = newTop + "px";
        panel.style.right = "auto";
        panel.style.bottom = "auto";
    });

    document.addEventListener("mouseup", function () {
        if (window.gcsPanelState.isDragging) {
            window.gcsPanelState.isDragging = false;
            // 이동이 끝나면 다시 지도의 카메라 회전 조작을 활성화합니다.
            if (window.viewer && window.viewer.scene) {
                window.viewer.scene.screenSpaceCameraController.enableInputs = true;
            }
        }
    });
});

function executeAddressSearch(query) {
    const resultList = document.getElementById("vworld_search_result_list");

    // [1단계] 먼저 지번 주소(PARCEL) 규격으로 안전하게 정밀 타겟팅 요청을 보냅니다.
    $.ajax({
        type: "get",
        url: "https://api.vworld.kr/req/search",
        data: {
            key: VWORLD_KEY,
            domain: "http://127.0.0.1:5500", // 로컬 주소 고정
            service: "address",
            version: "2.0",
            request: "getcoord",
            address: query,
            crs: "EPSG:4326",
            type: "PARCEL", // ◀ 브이월드 정식 지번 주소 표준 규격 지정
            format: "json"
        },
        dataType: "jsonp",
        jsonpCallback: "parseParcelSearchResponse",
        success: function (data) {
            console.log("🔍 [디버깅] 1차 지번 주소 응답:", data);

            if (data && data.response && data.response.status === "OK" && data.response.result) {
                // 지번 주소 결과가 성공적으로 존재하면 화면에 렌더링
                processAddressResult(data, query);
            } else {
                // ◀ [핵심 우회] 지번 결과가 없으면, 서버 규격을 도로명(ROAD)으로 바꾸어 2차 정밀 추적을 가동합니다.
                executeRoadAddressSearch(query);
            }
        },
        error: function () {
            // 통신 에러 발생 시에도 끊기지 않고 도로명 검색으로 토스합니다.
            executeRoadAddressSearch(query);
        }
    });
}

            service: "address",
            version: "2.0",
            // ◀ [정밀 교정 1] 유효 범위 오류 해결을 위해 정식 오퍼레이션 명칭인 Refine으로 교정

                processAddressResult(data, query);

                executeRoadAddressSearch(query);
            }
        },
        error: function () {

        }
    });
}

    $.ajax({
        type: "get",

            request: "getcoord",
            address: query,

            type: "ROAD", // 도로명 주소 분류 지정
            format: "json"
        },

// 📋 [파싱 구조 붕괴 원천 박멸본] Geocoder 2.0 표준 출력 포맷 매핑기
// =========================================================================
function processAddressResult(data, query) {
    let addrItems = [];

    // 🎯 브이월드 Geocoder 2.0 응답 객체의 모든 레이어 구조를 전수 분해합니다.
    if (data && data.response && data.response.result) {
        let res = data.response.result;

        // [핵심 교정] result 자체가 배열 형태로 넘어오는 브이월드 내부 특성을 가드링합니다.
        if (Array.isArray(res) && res.length > 0) {
            res = res[0]; // 첫 번째 결과 객체 강제 타겟팅
        } else if (res.items && Array.isArray(res.items) && res.items.length > 0) {
            res = res.items[0];
        }

        // 추출된 인프라 객체 내부에 좌표(point) 데이터가 정상 상주하는지 체크합니다.
        if (res && res.point && res.point.x && res.point.y) {
            let refinedText = query;

            // 정제된 명칭 구조 문자열 결합
            if (data.response.refined && data.response.refined.text) {
                refinedText = data.response.refined.text;
            } else if (res.structure && res.structure.full) {
                refinedText = res.structure.full;
            } else if (res.address && (res.address.road || res.address.parcel)) {
                refinedText = res.address.road || res.address.parcel;
            }

            addrItems = [{
                title: refinedText,
                address: { parcel: refinedText },
                point: { x: parseFloat(res.point.x), y: parseFloat(res.point.y) } // x(경도), y(위도) 추출 완수
            }];
        }
    }

    // 최종 교정 가드가 완료된 주소 엔티티 명단을 드롭다운 목록 UI로 전송합니다.
    if (addrItems.length > 0) {
        renderSearchResults(addrItems);
    } else {
        document.getElementById("vworld_search_result_list").innerHTML =
            `<li style="padding: 12px; color: #e74c3c; font-size: 12px; text-align: center;">주소 데이터는 수신되었으나 내부 좌표 파싱에 실패했습니다.</li>`;
    }
}

// 🏢 [건물 상세 정보 인프라] 브이월드 데이터 API 2.0 건축물 속성 추적 엔진
// =========================================================================

// 🚀 [Step 1] 사용자가 입력한 건물명으로 검색을 시작하는 게이트웨이 함수
function executeIntegratedSearch() {
    const query = document.getElementById("vworld_search_input").value.trim();
    const resultList = document.getElementById("vworld_search_result_list");

    if (!query) { alert("검색할 건물 명칭을 입력해 주세요."); return; }

    resultList.innerHTML = `<li style="padding: 12px; color: #bdc3c7; font-size: 12px; text-align: center;">건물 정보 탐색 중...</li>`;
    resultList.style.display = "block";

    // 브이월드 정식 PLACE(명칭) API를 호출하여 건물의 기본 위경도 좌표를 먼저 확보합니다.
    $.ajax({
        type: "get",
        url: "https://api.vworld.kr/req/search",
        data: {
            key: VWORLD_KEY,
            domain: "http://127.0.0.1:5500", // 로컬 호스트 고정
            service: "search",
            version: "2.0",
            request: "search",
            query: query,
            type: "PLACE",
            format: "json"
        },
        dataType: "jsonp",
        jsonpCallback: "parsePoiSearchResponse",
        success: function (data) {
            if (data.response && data.response.status === "OK" && data.response.result && data.response.result.items) {
                // 확보된 명칭 리스트를 화면에 인쇄하기 위해 전달
                renderSearchResults(data.response.result.items);
            } else {
                // 명칭 검색 결과가 전혀 없다면 일반 주소 변환(Geocoder) 엔진으로 예외 처리 연동
                if (typeof executeAddressSearch === "function") {
                    executeAddressSearch(query);
                } else {
                    resultList.innerHTML = `<li style="padding: 12px; color: #e74c3c; font-size: 12px; text-align: center;">검색 결과가 없습니다.</li>`;
                }
            }
        },
        error: function () {
            resultList.innerHTML = `<li style="padding: 12px; color: #e74c3c; font-size: 12px; text-align: center;">통신 중 에러가 발생했습니다.</li>`;
        }
    });
}

// 📋 [2D 평면 디버깅 디폴트 고정본] 검색 결과 리스트 UI 및 카메라 제어 엔진
// =========================================================================
function renderSearchResults(items) {
    const resultList = document.getElementById("vworld_search_result_list");
    if (!resultList) return;
    resultList.innerHTML = "";

    items.forEach((item) => {
        const title = item.title;
        const subAddress = item.address ? (item.address.road || item.address.parcel || "") : "";

        // 브이월드 서버 수신 데이터 좌표 실수 변환
        const lon = parseFloat(item.point.x);
        const lat = parseFloat(item.point.y);

        const li = document.createElement("li");
        li.style.cssText = `
            padding: 10px 12px; border-bottom: 1px solid #34495e;
            cursor: pointer; transition: background 0.2s;
        `;
        li.innerHTML = `
            <div style="font-size: 13px; color: #fff; font-weight: bold; overflow: hidden; text-overflow: ellipsis; white-space: nowrap;">${title}</div>
            <div style="font-size: 11px; color: #95a5a6; margin-top: 3px; overflow: hidden; text-overflow: ellipsis; white-space: nowrap;">${subAddress}</div>
        `;

        // 마우스 오버 시 UI 하이라이트 효과
        li.addEventListener("mouseenter", () => li.style.background = "#2e86de");
        li.addEventListener("mouseleave", () => li.style.background = "transparent");

        // 🎯 검색 항목 리스트 클릭 시 2D 수직 탑다운 고정 비행 기동
        li.addEventListener("click", function () {
            if (!ws3d || !ws3d.viewer) return;
            const viewer = ws3d.viewer;

            // [대시보드 UI 연동] 카메라가 2D로 수직 하강하므로, 중앙 패널의 버튼 활성화 상태도 함께 2D로 동기화합니다.
            window.isDrawingMode = true; // 드로잉 드로잉 모드 트리거 온
            const btn2d = document.getElementById("btn_2d");
            const btn3d = document.getElementById("btn_3d");
            if (btn2d) btn2d.className = "btn-mode active";
            if (btn3d) btn3d.className = "btn-mode";

            // 기존 검색용 빨간점 핀 파괴 및 완전 리셋
            const oldSearchPoint = viewer.entities.getById("vworld_search_marker_pin");
            if (oldSearchPoint) { viewer.entities.remove(oldSearchPoint); }

            // 위·경도 타겟 지점(고도 Z값 = 0)에 붉은색 3D 점(Entity)을 정밀 도포합니다.
            viewer.entities.add({
                id: "vworld_search_marker_pin",
                position: Cesium.Cartesian3.fromDegrees(lon, lat, 0.0), // Z값 0 지표면 밀착 고수
                point: {
                    pixelSize: 18,
                    color: Cesium.Color.RED,
                    outlineColor: Cesium.Color.WHITE,
                    outlineWidth: 3,
                    disableDepthTestDistance: Number.POSITIVE_INFINITY
                },
                label: {
                    text: "🎯 " + title,
                    font: 'bold 12px sans-serif',
                    style: Cesium.LabelStyle.FILL_AND_OUTLINE,
                    fillColor: Cesium.Color.WHITE,
                    outlineColor: Cesium.Color.BLACK,
                    outlineWidth: 3,
                    verticalOrigin: Cesium.VerticalOrigin.BOTTOM,
                    pixelOffset: new Cesium.Cartesian2(0, -15),
                    disableDepthTestDistance: Number.POSITIVE_INFINITY
                }
            });

            // ✈️ [핵심 요청 조치] 3D 사선 사각 뷰를 완전히 배제하고, 
            // 정중앙 수직 위(Pitch: -90도) 상공 고도 250m 위치에서 정수리를 정밀 타겟팅 조준하도록 설계합니다.
            const cameraDestination2D = Cesium.Cartesian3.fromDegrees(lon, lat, 250.0);

            viewer.camera.flyTo({
                destination: cameraDestination2D,
                orientation: {
                    heading: Cesium.Math.toRadians(0.0),
                    pitch: Cesium.Math.toRadians(-90.0), // ◀ 정수리를 정확히 내려다보는 수직 2D 평면 설계 가각 고정
                    roll: 0.0
                },
                duration: 1.2, // 1.2초 만에 즉시 축 정렬 강제 진입
                complete: function () {
                    // 비행 완료 즉시 캔버스 리사이저를 강제 작동시켜 미세 왜곡 버그를 소멸합니다.
                    if (window.triggerCesiumResize) window.triggerCesiumResize();
                }
            });

            resultList.style.display = "none"; // 팝업 닫기

            // 백그라운드에서 현재 클릭지의 위경도 좌표 기준 데이터대장 파싱 스레드 호출 연동
            if (typeof fetchBuildingDetailInformation === "function") {
                fetchBuildingDetailInformation(lon, lat, title);
            }
        });

        resultList.appendChild(li);
    });
}

function processFinalBuildingData(props, lon, lat, isPinAction) {
    if (isPinAction) {

            window.executeRenderFixedPopup(lon, lat, props);
        }
    } else {
        window.vworldHoverTooltip.props = props;
        const tooltip = window.vworldHoverTooltip.element;
        
        // 🎯 [핵심 요청 사항 반영] 건물상세팝업 체크박스가 켜져있을(ON) 때만 지도 화면 마우스 머리맡에 팝업창을 인쇄합니다!
        // 꺼져있을(OFF) 때는 코드가 이 블록을 패스하므로 화면에 팝업창이 전혀 출력되지 않아 깨끗해집니다.
        if (tooltip && window.isBuildingPopupEngineActive) {
            tooltip.style.display = "block !important";
            tooltip.style.setProperty("display", "block", "important");
            
            tooltip.innerHTML = `

        } else if (tooltip) {
            // 체크박스가 꺼져있다면 가시성을 즉시 차단 투명화
            tooltip.style.display = "none";
        }
    }

    // 🟢 [상시 유지 1] 팝업창 체크박스 상태와 전혀 무관하게, 왼쪽 대시보드(Map Board) UI 명세 주입은 항상 100% 정상 작동합니다!
    if(typeof updateBuildingDashboardUI === "function") {

    if (!isPinAction && typeof window.drawFireZoneOnGround === "function") {
        window.drawFireZoneOnGround(lon, lat); 
    }

// 🔄 [세슘 카메라 포스트 렌더링 스크린 투영 엔진 - 크래시 완벽 보호 버전]
function updatePopupPositionsOnScreen() {
    if (typeof ws3d === "undefined" || !ws3d || !ws3d.viewer || !ws3d.viewer.scene || !ws3d.viewer.scene.canvas) return;

    if (!window.vworldFixedPopups || !Array.isArray(window.vworldFixedPopups)) {
        window.vworldFixedPopups = [];
        return;
    }

    const scene = ws3d.viewer.scene;

    // A. 🎯 [x오류 박멸 핵심 가드] 실시간 자동 호버 툴팁 추적
    if (window.vworldHoverTooltip && window.vworldHoverTooltip.element && window.vworldHoverTooltip.position) {
        if (window.vworldHoverTooltip.element.style.display === "block") {
            const canvasPos = Cesium.SceneTransforms.wgs84ToWindowCoordinates(scene, window.vworldHoverTooltip.position);
            // canvasPos 자체가 undefined 이거나 물리 x, y가 파싱되지 않았다면 연산을 스킵하여 크래시를 박멸합니다.
            if (Cesium.defined(canvasPos) && canvasPos.x !== undefined && canvasPos.y !== undefined) {
                window.vworldHoverTooltip.element.style.left = canvasPos.x + "px";
                window.vworldHoverTooltip.element.style.top = (canvasPos.y - 12) + "px";
            }
        }
    }
    // B. 멀티 박제 고정핀 팝업들 실시간 투영 추적
    window.vworldFixedPopups.forEach((popup) => {
        if (!popup || !popup.position || !popup.element) return;
        const canvasPosition = Cesium.SceneTransforms.wgs84ToWindowCoordinates(scene, popup.position);
        if (Cesium.defined(canvasPosition) && canvasPosition.x !== undefined && canvasPosition.y !== undefined) {
            popup.element.style.display = "block";
            popup.element.style.left = canvasPosition.x + "px";
            popup.element.style.top = (canvasPosition.y - 12) + "px";
        } else {
            popup.element.style.display = "none";
        }
    });
}

// 🔄 [세슘 카메라 포스트 렌더링 스크린 투영 엔진 - 크래시 완벽 보호 버젼]
function updatePopupPositionsOnScreen() {
    // 🎯 [철통 가드 1] 세슘 엔진 인스턴스가 완벽히 성립되지 않았다면 무조건 즉시 실행을 거부합니다.
    if (typeof ws3d === "undefined" || !ws3d || !ws3d.viewer || !ws3d.viewer.scene || !ws3d.viewer.scene.canvas) return;

    // 🎯 [철통 가드 2] vworldFixedPopups 가 어떤 원인에 의해 깨졌다면 실시간으로 강제 복구해 렌더링 중단을 예방합니다.
    if (!window.vworldFixedPopups || !Array.isArray(window.vworldFixedPopups)) {
        window.vworldFixedPopups = [];
        return;
    }

    const scene = ws3d.viewer.scene;

    // A. 🎯 [x오류 박멸 핵심] 실시간 자동 호버 툴팁 추적
    if (window.vworldHoverTooltip && window.vworldHoverTooltip.element && window.vworldHoverTooltip.position) {
        if (window.vworldHoverTooltip.element.style.display === "block") {
            const canvasPos = Cesium.SceneTransforms.wgs84ToWindowCoordinates(scene, window.vworldHoverTooltip.position);

            // ◀ canvasPos 자체가 undefined 이거나 물리 x, y가 파싱되지 않았다면 연산을 스킵하여 크래시를 박멸합니다.
            if (Cesium.defined(canvasPos) && canvasPos.x !== undefined && canvasPos.y !== undefined) {
                window.vworldHoverTooltip.element.style.left = canvasPos.x + "px";
                window.vworldHoverTooltip.element.style.top = (canvasPos.y - 12) + "px";
            }
        }
    }

    // B. 멀티 박제 고정핀 팝업들 실시간 투영 추적
    window.vworldFixedPopups.forEach((popup) => {
        if (!popup || !popup.position || !popup.element) return;

        const canvasPosition = Cesium.SceneTransforms.wgs84ToWindowCoordinates(scene, popup.position);
        if (Cesium.defined(canvasPosition) && canvasPosition.x !== undefined && canvasPosition.y !== undefined) {
            popup.element.style.display = "block";
            popup.element.style.left = canvasPosition.x + "px";
            popup.element.style.top = (canvasPosition.y - 12) + "px";
        } else {
            popup.element.style.display = "none";
        }
    });
}

/**
 * 🗺️ [연속지적도 하이브리드 토글 인터페이스 최종 마스터본]
 * 베이스 위성 맵을 건드리지 않고, 전역에 보관된 나의 지적도 인스턴스만 정밀 조준하여 제거합니다.
 */
window.handleJiJeokIntegratedToggle = function (checkbox) {
    if (!map) {
        console.warn("⚠️ 브이월드 지도 엔진이 아직 로드되지 않았습니다.");
        return;
    }

    if (checkbox.checked) {
        // 1. 체크 가동 시 지적도 타일 렌더러 기동
        if (typeof wfsSample === "function") {
            wfsSample(); 
            console.log("🟢 [GCS 대시보드] 연속지적도 오버레이 및 클릭 피킹 엔진 ON");
        }
    } else {
        // 2. 🚨 [화이트맵 증상 완벽 해결] 베이스 맵을 파괴하지 않고 연속지적도 레이어만 쏙 골라서 지웁니다.
        if (window.viewer && window.viewer.imageryLayers) {
            const layers = window.viewer.imageryLayers;
            
            // 전역 변수에 백업해 둔 나의 지적도 인스턴스가 레이어 풀에 존재하는지 확인
            if (window.vworldJijeokTileInstance && layers.contains(window.vworldJijeokTileInstance)) {
                try {
                    // 전체를 숨기지 않고 해당 지적도 레이어 딱 '하나만' 풀에서 삭제합니다.
                    layers.remove(window.vworldJijeokTileInstance);
                    window.vworldJijeokTileInstance = null;
                    console.log("🧹 [정밀 청소] 전역 인스턴스 매칭을 통한 지적도 레이어 단독 소멸");
                } catch (err) { }
            } else {
                // 만약 인스턴스 매칭이 안 될 경우를 대비한 2차 안전 가드 루프 (역순 순회)
                for (let i = layers.length - 1; i >= 0; i--) {
                    const layer = layers.get(i);
                    // 레이어 내부의 레이어 레이블 명칭이나 빌더 옵션 주소에 지적도 키워드(cbnd)가 묻어있는지 체크
                    if (layer && layer.imageryProvider && layer.imageryProvider._layers) {
                        if (layer.imageryProvider._layers.includes("cbnd")) {
                            layers.remove(layer); // 지적도 레이어만 골라 삭제
                            console.log("🧹 [루프 청소] 레이어 내부 레이어 명칭 역추적을 통한 지적도 삭제");
                        }
                    }
                }
            }
        } else if (map && window.vworldJijeokTileInstance) {
            // 2D 맵 환경일 때의 예외 처리 가드
            try {
                map.getLayers().remove(window.vworldJijeokTileInstance);
                window.vworldJijeokTileInstance = null;
            } catch (e) { }
        }

        // 3. 마우스 클릭 WFS 조회 이벤트 리스너 해제 및 전역 상태 가드 리셋
        if (typeof wfsEvent === "function" && map.onClick && map.onClick.removeEventListener) {
            try {
                map.onClick.removeEventListener(wfsEvent);
                if (typeof chk !== "undefined") chk = 0; 
            } catch (e) { }
        }
        
        // 4. 대시보드 UI 결과창 텍스트 초기화 동기화
        $('#resultDiv').html("<li style='color:#7f8c8d;'>• 연속지적도 레이어가 해제되었습니다.</li>");
        console.log("🧹 [GCS 대시보드] 연속지적도 완전 소멸 및 화이트맵 증상 복구 완수");
    }
};
/**
 * [추가 백업 가드] 지적도 레이어를 타일 풀에서 찾아 완벽하게 청소하는 서브 클리너 함수
 */
function clearJiJeokTileLayers() {
    try {
        if (window.viewer && window.viewer.imageryLayers) {
            const layers = window.viewer.imageryLayers;
            for (let i = layers.length - 1; i >= 0; i--) {
                const layer = layers.get(i);
                // 지적도 레이어 프로바이더 특성 명칭 분석을 통한 정밀 필터 타겟팅 제거
                if (layer && layer.imageryProvider && typeof layer.imageryProvider.getTileCredits === "function") {
                    if (layer.show !== undefined) layer.show = false;
                }
            }
        }
        $('#resultDiv').html("<li style='color:#7f8c8d;'>연속지적도 레이어가 해제되었습니다.</li>");
    } catch(e) {
        $('#resultDiv').html("<li>레이어 해제 연산 완료</li>");
    }
}
