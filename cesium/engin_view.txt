/**
 * 💥 [원본 복원] 브이월드 WFS 전역 검색 결과 수신 처리 함수 (필지 속성 분석 파이프라인)
 */
window.parseResponse = function (data) {
    if (!data) {
        alert("검색결과가 없습니다.");
        return;
    }
    var properties = null;
    if (data.features) {
        if (Array.isArray(data.features) && data.features.length > 0) {
            properties = data.features[0].properties; // 첫 번째 검색 객체 인덱싱 복원
        } else if (data.features.properties) {
            properties = data.features.properties;
        }
    }
    if (!properties) return;

    // 도로명/지번에 따른 원본 필지 상세 명세 정보 리스트 HTML 빌딩
    var resultHtml = "";
    resultHtml += "<li> - pnu :   " + (properties.pnu || "-") + "</li>";
    resultHtml += "<li> - jibun : " + (properties.jibun || "-") + "</li>";
    resultHtml += "<li> - addr :  " + (properties.addr || "-") + "</li>";
    resultHtml += "<li> - jiga :  " + (properties.jiga || "-") + "</li>";

    // 사이드바 내부 결과 출력 div 엘리먼트 동기화 가드
    if (jQuery('#resultDiv').length > 0) {
        jQuery('#resultDiv').html(resultHtml);
    }
};

/**
 * 💥 [최종 마스터 완치본] 필수 category 파라미터 주입형 지능형 검색 라우터 엔진
 */
window.executeGcsSearch = function() {
    var searchInput = document.getElementById("gcs_search_keyword");
    var query = searchInput ? searchInput.value.trim() : "";
    var resultList = document.getElementById("resultDiv"); 
    var popupLayer = document.getElementById("gcs_search_layer_popup");

    if (!query) { alert("검색어를 입력하세요."); return; }
    if (resultList) {
        resultList.innerHTML = "<li style='padding: 12px; color: #ccc; text-align: center; list-style:none;'>탐색 중...</li>";
    }
    if (popupLayer) popupLayer.style.display = "block"; 

    // A. 1단계: 명칭(PLACE)과 주소(ADDRESS) 1차 자동 분류 가드
    var isAddress = false;
    if (query.indexOf("로") !== -1 || query.indexOf("길") !== -1 || query.indexOf("동") !== -1 || 
        query.indexOf("읍") !== -1 || query.indexOf("면") !== -1 || query.indexOf("리") !== -1 || 
        query.indexOf("가") !== -1 || /[0-9]/.test(query)) {
        isAddress = true; 
    }

    var searchType = isAddress ? "ADDRESS" : "PLACE";
    
    // 💥 [콘솔 에러 완치 핵심부] 주소(ADDRESS) 채널 구동 시 브이월드가 필수 요구하는 category 변수 자동 연산 세팅
    var categoryParam = "";
    if (searchType === "ADDRESS") {
        // '로' 또는 '길' 혹은 숫자가 단독 조합된 형태면 도로명 주소(road) 확정, 그 외 행정구역은 지번(parcel) 확정
        if (query.indexOf("로") !== -1 || query.indexOf("길") !== -1) {
            categoryParam = "road"; 
        } else {
            categoryParam = "parcel";
        }
    }

    console.log("📡 [GCS 지능형 라우터기] 필수 명세 결착 파라미터 -> 유형: " + searchType + " | 카테고리: " + (categoryParam || "N/A") + " | 검색어: " + query);

    // 개발자님이 완벽하게 성공하셨던 순정 jQuery.ajax 프로토콜 명세에 필수 조건 결착 전송
    var requestData = {
        key: GCS_VWORLD_API_KEY, 
        domain: "127.0.0.1",     
        service: "search", 
        version: "2.0", 
        request: "search", 
        query: query, 
        type: searchType,
        format: "json"
    };

    // 💥 주소 검색인 경우 필수 카테고리 속성을 쿼리 셋에 다이렉트 주입
    if (searchType === "ADDRESS") {
        requestData.category = categoryParam;
    }

    jQuery.ajax({
        type: "get",
        url: "https://api.vworld.kr/req/search",
        data: requestData,
        dataType: "jsonp",
        jsonpCallback: "parsePoiSearchResponse", // 브이월드 서버 전용 무선 보안 콜백 명세 래칭
        success: function (data) {
            if (data && data.response && data.response.status === "NOT_FOUND") {
                if (resultList) resultList.innerHTML = "<li style='padding: 12px; color: red; text-align: center; list-style:none;'>결과가 없습니다.</li>";
            }
        }
    });
};


/**
 * 💥 [최종 완치 & 콘솔 출력 주입] 브이월드 JSONP 서버 수신 인터페이스 통로 파이프라인
 */
window.parsePoiSearchResponse = function(data) {
    var resultList = document.getElementById("resultDiv");
    if (!resultList) return;

    // 📡 [요청 반영] 브이월드 서버에서 날아온 순정 데이터 객체를 브라우저 콘솔창에 그대로 표출
    console.log("======================================================");
    console.log("📡 [GCS 수신부] 브이월드 서버 응답 원본 수신 데이터 (Raw Data):");
    console.log(data);
    console.log("======================================================");

    if (data && data.response && data.response.status === "OK" && data.response.result) {
        var res = data.response.result;
        
        // 명칭 필드(items)와 도로명/지번 주소용 필드(juso 또는 item)를 전수 바인딩
        var rawData = res.items || res.juso || res.item;
        
        if (rawData) {
            window.renderGcsSearchResults(rawData);
            return;
        }
    }
    
    if (resultList) {
        resultList.innerHTML = "<li style='padding: 12px; color: red; text-align: center; list-style:none;'>결과가 없습니다.</li>";
    }
};

//**
//  * 💥 [최종 마스터 완치본] 이름 없음 현상을 해결하고 도로명 주소를 정밀 피딩하는 통합 렌더러
// 

window.renderGcsSearchResults = function(items) {
    var resultList = document.getElementById("resultDiv");
    if (!resultList) return;
    resultList.innerHTML = "";

    var itemsArray = [];
    if (Array.isArray(items)) {
        itemsArray = items;
    } else if (items && typeof items === "object") {
        itemsArray.push(items);
    }

    console.log("📋 [GCS 렌더러] 화면에 바인딩 처리될 최종 리스트 구성 요소를 출력합니다 (Array):");
    console.log(itemsArray);
    console.log("======================================================");

    itemsArray.forEach(function(item) {
        if (!item) return;
        
        // 💥 [이름 없음 현상 정밀 완치 파트] 
        // 1. 일반 명칭(title) 분석 -> 2. 콘솔에서 검증된 정식 도로명 주소(address.road) 구조 순차 바인딩
        var displayTitle = item.title || item.text || "이름 없음";
        
        if (displayTitle === "이름 없음" && item.address) {
            if (typeof item.address === "string") {
                displayTitle = item.address; // 주소 문자열이 통째로 들어온 경우 직렬화
            } else if (item.address.road) {
                displayTitle = item.address.road; // 콘솔 덤프에 찍힌 정식 도로명 매핑
            } else if (item.address.parcel) {
                displayTitle = item.address.parcel; // 지번 주소 매핑 백업
            }
        }
        
        // 하이브리드 위경도 좌표축 정밀 분해 해독 가드
        var targetPoint = null;
        if (item.point) {
            targetPoint = item.point;
        } else if (item.geometry && item.geometry.point) {
            targetPoint = item.geometry.point;
        }
        
        if (!targetPoint) return;
        var lon = parseFloat(targetPoint.x); 
        var lat = parseFloat(targetPoint.y);
        if (isNaN(lon) || isNaN(lat)) return;
        
        // 관제 레이아웃 UI 결착
        var li = document.createElement("li");
        li.style.cssText = "padding: 10px; border-bottom: 1px solid #34495e; color:#fff; cursor:pointer; list-style:none; text-align:left; transition:0.2s;";
        
        // 💥 타이틀 영역에 '이름 없음' 대신 정제된 도로명 주소가 선명하게 표출되도록 인젝션
        li.innerHTML = "<div><strong style='color: #00ffcc;'>📍 " + displayTitle + "</strong></div>" + 
                       "<div style='font-size:10px; color:#bdc3c7; margin-top:3px;'>좌표 - X: " + lon.toFixed(4) + ", Y: " + lat.toFixed(4) + "</div>";
        
        li.onmouseover = function() { this.style.background = "rgba(26, 188, 156, 0.25)"; };
        li.onmouseout = function() { this.style.background = "transparent"; };

        // 리스트 항목 선택(클릭) 시 빨간색 마커 생성 및 고도 250m 수직 탑뷰(-90도) 카메라 flyTo 기동 인터록
        li.addEventListener("click", function () {
            if (typeof ws3d === "undefined" || !ws3d || !ws3d.viewer) return;
            
            ws3d.viewer.camera.lookAtTransform(Cesium.Matrix4.IDENTITY);
            window.isDrawingMode = true;
            
            var oldSearchPoint = ws3d.viewer.entities.getById("vworld_search_marker_pin");
            if (oldSearchPoint) { 
                ws3d.viewer.entities.remove(oldSearchPoint); 
            }

            // 빨간색 마커 핀 결착 생성
            ws3d.viewer.entities.add({
                id: "vworld_search_marker_pin", 
                position: Cesium.Cartesian3.fromDegrees(lon, lat, 0.0),
                point: { 
                    pixelSize: 18, 
                    color: Cesium.Color.RED, 
                    outlineColor: Cesium.Color.WHITE, 
                    outlineWidth: 3, 
                    disableDepthTestDistance: Number.POSITIVE_INFINITY 
                }
            });

            // 검증 완료된 수직 수하 -90도 조감 탑뷰 구도 flyTo 기동
            ws3d.viewer.camera.flyTo({
                destination: Cesium.Cartesian3.fromDegrees(lon, lat, 250.0),
                orientation: { heading: Cesium.Math.toRadians(0.0), pitch: Cesium.Math.toRadians(-90.0), roll: 0.0 },
                duration: 1.2
            });

            var popupLayer = document.getElementById("gcs_search_layer_popup");
            if (popupLayer) popupLayer.style.display = "none";
        });
        
        resultList.appendChild(li);
    });

    if (resultList.children.length === 0) {
        resultList.innerHTML = "<li style='padding: 12px; color: red; text-align: center; list-style:none;'>결과가 없습니다.</li>";
    }
};

/**
 * 세슘 3D 웹글 마우스 인터랙션 바인딩
 */
vw.ws3dInitCallBack = function () {
    window.viewer = ws3d.viewer;
    const scene = ws3d.viewer.scene;
    const handler = new Cesium.ScreenSpaceEventHandler(scene.canvas);
    scene.canvas.addEventListener("contextmenu", function (e) { e.preventDefault(); e.stopPropagation(); return false; }, { passive: false });

    let lastHoveredFeature = undefined;
    handler.setInputAction(function (moveEvent) {
        if (!moveEvent.endPosition) return;
        const pickedFeature = scene.pick(moveEvent.endPosition);
        if (Cesium.defined(pickedFeature) && pickedFeature.primitive && !(pickedFeature.primitive instanceof Cesium.Globe)) {
            document.body.style.cursor = "pointer";
            if (pickedFeature !== lastHoveredFeature) {
                if (lastHoveredFeature) { try { lastHoveredFeature.color = Cesium.Color.WHITE; } catch (e) { } }
                lastHoveredFeature = pickedFeature;
                try { pickedFeature.color = Cesium.Color.YELLOW.withAlpha(0.40); } catch (e) { }
            }
        } else {
            if (lastHoveredFeature) { try { lastHoveredFeature.color = Cesium.Color.WHITE; } catch (e) { } lastHoveredFeature = undefined; }
            document.body.style.cursor = "default";
        }
    }, Cesium.ScreenSpaceEventType.MOUSE_MOVE);

    handler.setInputAction(function (rightClickEvent) {
        if (!rightClickEvent.position) return;
        const contextMousePosition = rightClickEvent.position;
        let pickedCartesian = scene.pickPosition(contextMousePosition);
        if (!Cesium.defined(pickedCartesian)) pickedCartesian = scene.globe.pick(scene.camera.getPickRay(contextMousePosition), scene);
        if (!Cesium.defined(pickedCartesian)) pickedCartesian = scene.camera.pickEllipsoid(contextMousePosition, scene.globe.ellipsoid);

        if (Cesium.defined(pickedCartesian)) {
            const cartographic = Cesium.Cartographic.fromCartesian(pickedCartesian);
            var rawLon = Cesium.Math.toDegrees(cartographic.longitude);
            var rawLat = Cesium.Math.toDegrees(cartographic.latitude);
            if (!isNaN(rawLon) && !isNaN(rawLat) && rawLon !== 0 && rawLat !== 0) {
                window.gcsLastRightClickLon = rawLon; window.gcsLastRightClickLat = rawLat;
                const ctxMenu = document.getElementById("gcs_context_menu");
                if (ctxMenu) {
                    ctxMenu.style.left = contextMousePosition.x + "px"; ctxMenu.style.top = contextMousePosition.y + "px"; ctxMenu.style.display = "block";
                    const menuItems = ctxMenu.children;
                    for (let i = 0; i < menuItems.length; i++) {
                        let item = menuItems[i];
                        item.style.cursor = "pointer";
                        if (item.textContent.includes("홈 설정")) item.onclick = function (e) { e.stopPropagation(); window.triggerGcsCommand("home"); ctxMenu.style.display = "none"; };
                        else if (item.textContent.includes("경유지 추가")) item.onclick = function (e) { e.stopPropagation(); window.triggerGcsCommand("waypoint"); ctxMenu.style.display = "none"; };
                        else if (item.textContent.includes("목적지 설정")) item.onclick = function (e) { e.stopPropagation(); window.triggerGcsCommand("destination"); ctxMenu.style.display = "none"; };
                    }
                }
            }
        }
    }, Cesium.ScreenSpaceEventType.RIGHT_CLICK);

    scene.canvas.addEventListener("mousedown", function (e) { var m = document.getElementById("gcs_context_menu"); if (m && e.button === 0) m.style.display = "none"; });
    if (typeof window.initDisasterWMSLayer === "function") window.initDisasterWMSLayer();
};

window.switchGcsMode = function (mode) {
    if (!ws3d || !ws3d.viewer) return;
    const canvas = ws3d.viewer.scene.canvas;
    const ray = ws3d.viewer.scene.camera.getPickRay(new Cesium.Cartesian2(canvas.clientWidth / 2, canvas.clientHeight / 2));
    const targetCartesian = ws3d.viewer.scene.globe.pick(ray, ws3d.viewer.scene);
    let centerLon = 126.9780, centerLat = 37.5665, currentHeight = ws3d.viewer.camera.positionCartographic.height;
    if (Cesium.defined(targetCartesian)) {
        const carto = Cesium.Cartographic.fromCartesian(targetCartesian);
        centerLon = Cesium.Math.toDegrees(carto.longitude); centerLat = Cesium.Math.toDegrees(carto.latitude);
    }
    ws3d.viewer.camera.lookAtTransform(Cesium.Matrix4.IDENTITY);
    if (mode === '2D') {
        window.isDrawingMode = true; document.getElementById("btn_2d").className = "btn-mode active"; document.getElementById("btn_3d").className = "btn-mode";
        ws3d.viewer.camera.flyTo({ destination: Cesium.Cartesian3.fromDegrees(centerLon, centerLat, currentHeight > 0 ? currentHeight : 350.0), orientation: { heading: Cesium.Math.toRadians(0.0), pitch: Cesium.Math.toRadians(-90.0), roll: 0.0 }, duration: 1.0 });
    } else {
        window.isDrawingMode = false; document.getElementById("btn_2d").className = "btn-mode"; document.getElementById("btn_3d").className = "btn-mode active";
        ws3d.viewer.camera.flyTo({ destination: Cesium.Cartesian3.fromDegrees(centerLon, centerLat - 0.002, currentHeight > 0 ? currentHeight : 300.0), orientation: { heading: Cesium.Math.toRadians(0.0), pitch: Cesium.Math.toRadians(-45.0), roll: 0.0 }, duration: 1.0 });
    }
};

    container.innerHTML = "";

 * 🎯 [우클릭 명령 분기] 홈, 경유지, 목적지 좌표 래칭 연산 엔진

window.triggerGcsCommand = function (actionType) {
    if (typeof ws3d === "undefined" || !ws3d || !ws3d.viewer) return;
    var rawHeight = Math.round(ws3d.viewer.camera.positionCartographic.height);
    var halvedHeight = Math.round((rawHeight <= 0 ? 300 : rawHeight) * 0.5);
    var numLon = parseFloat(window.gcsLastRightClickLon), numLat = parseFloat(window.gcsLastRightClickLat);
    if (isNaN(numLon) || isNaN(numLat) || numLon === 0) return;

    if (actionType === "home") {
        var hData = { id: "Home", lon: numLon, lat: numLat, alt: halvedHeight };
        if (window.waypointList.length > 0 && window.waypointList[0].id === "Home") window.waypointList[0] = hData;
        else window.waypointList.unshift(hData);
    } else if (actionType === "destination") {
        var dData = { id: "Destination", lon: numLon, lat: numLat, alt: halvedHeight };
        if (window.waypointList.length > 0 && window.waypointList[window.waypointList.length - 1].id === "Destination") {
            window.waypointList[window.waypointList.length - 1] = dData;
        } else {
            window.waypointList.push(dData); // window.waypointList 누수 버그 교정 완료
        }
    } else if (actionType === "waypoint") {
        var idx = window.waypointList.length;
        if (window.waypointList.length > 0 && window.waypointList[window.waypointList.length - 1].id === "Destination") idx = window.waypointList.length - 1;
        window.waypointList.splice(idx, 0, { id: "WP_TEMP", lon: numLon, lat: numLat, alt: halvedHeight });
    }
    
    // 노드 네이밍 정렬 및 데이터 유실 원천 차단 파이프라인
    var cleanList = []; var wpCounter = 1;
    window.waypointList.forEach(function (wp, idx) {
        var lbl = (idx === 0) ? "Home" : (idx === window.waypointList.length - 1 && window.waypointList.length > 1 && wp.id === "Destination") ? "Destination" : "WP_" + wpCounter++;
        cleanList.push({ id: lbl, lon: wp.lon, lat: wp.lat, alt: wp.alt });
    });
    window.waypointList = cleanList;
    window.rebuildGcsPipelineAndLabels();
};

/**
 * 🛸 [삼각기둥 연동] 3D 입체 에어돔 공역 및 순항 가이드 polyline 실시간 동기화 빌더
 */
window.rebuildGcsPipelineAndLabels = function () {
    if (typeof ws3d === "undefined" || !ws3d || !ws3d.viewer) return;
    window.gcsFireZoneEntities.forEach(function (e) { ws3d.viewer.entities.remove(e); });
    window.gcsFireZoneEntities = []; 
    var pipeLinePositions = [];

    window.waypointList.forEach(function (node) {
        var nodeAlt = parseFloat(node.alt) || 150.0;
        var upperAir = Cesium.Cartesian3.fromDegrees(node.lon, node.lat, nodeAlt);
        pipeLinePositions.push(upperAir);
        
        var nodeColor = (node.id === "Home") ? Cesium.Color.DARKORANGE : (node.id === "Destination") ? Cesium.Color.RED : Cesium.Color.CYAN;
        var trianglePos = [];
        for (var a = 0; a < 360; a += 120) { 
            var rad = Cesium.Math.toRadians(a); 
            trianglePos.push(node.lon + (0.00025 * Math.cos(rad)), node.lat + (0.00025 * Math.sin(rad))); 
        }

        // 3D 삼각기둥 공역 엔티티 드로잉
        window.gcsFireZoneEntities.push(ws3d.viewer.entities.add({ 
            polygon: { hierarchy: Cesium.Cartesian3.fromDegreesArray(trianglePos), height: 0.0, extrudedHeight: nodeAlt, material: nodeColor.withAlpha(0.32), outline: true, outlineColor: nodeColor.withAlpha(0.6), outlineWidth: 3 } 
        }));
        
        // 상공 고도 표시 레이블 드로잉
        window.gcsFireZoneEntities.push(ws3d.viewer.entities.add({ 
            position: Cesium.Cartesian3.fromDegrees(node.lon, node.lat, nodeAlt + 15.0), 
            label: { text: node.id + " (" + Math.round(nodeAlt) + "m)", font: 'bold 13px "Malgun Gothic", sans-serif', fillColor: nodeColor, outlineColor: Cesium.Color.BLACK, outlineWidth: 3, style: Cesium.LabelStyle.FILL_AND_OUTLINE, verticalOrigin: Cesium.VerticalOrigin.BOTTOM, disableDepthTestDistance: Number.POSITIVE_INFINITY } 
        }));
    });

    // 노드 간을 이어주는 녹색 발광 순항선(Polyline) 결착
    if (pipeLinePositions.length > 1) {
        window.gcsFireZoneEntities.push(ws3d.viewer.entities.add({ 
            polyline: { positions: pipeLinePositions, width: 5, material: new Cesium.PolylineGlowMaterialProperty({ glowPower: 0.3, color: Cesium.Color.SPRINGGREEN }), arcType: Cesium.ArcType.NONE } 
        }));
    }
    
    window.renderWaypointListUI(); 
    ws3d.viewer.scene.requestRender();
};

    }
