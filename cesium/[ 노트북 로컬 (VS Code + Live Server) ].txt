// 🚨 [fire.js 최종 마스터 - 파트 1: 브이월드 전용 프록시 터널링 우회 마스터본]

// 전역 스코프에서 레이어 인스턴스를 관리하기 위해 전역 변수로 선언합니다.
window.vworldDisasterLayerInstance = null;

/**
 * 🎨 [기능 1] 지도가 초기화된 후 도로명주소건물(lt_c_spbd) 레이어를 지도 위에 상시 표현합니다.
 * 🔴 [CORS & Tainted Canvas 완벽 해결] 브이월드 공식 보안 우회 프록시 관로를 강제로 태워
 * WebGL 캔버스 텍스처 추출 시 발생하는 모든 SecurityError를 원천 차단합니다.
 */
window.initDisasterWMSLayer = function () {
    if (typeof ws3d === "undefined" || !ws3d || !ws3d.viewer) {
        console.warn("⚠️ 세슘 엔진이 아직 로드되지 않아 레이어를 주입할 수 없습니다.");
        return;
    }

    try {
        // 🛑 중복 생성 방지 가드
        if (window.vworldDisasterLayerInstance) {
            ws3d.viewer.imageryLayers.remove(window.vworldDisasterLayerInstance);
            window.vworldDisasterLayerInstance = null;
        }

        // 🎯 브이월드 표준 WMS 이미지 프로바이더 규격 가동
        const disasterWmsProvider = new Cesium.WebMapServiceImageryProvider({
            url: "https://vworld.kr",
            layers: "lt_c_spbd",                  // 성공 명세서 표준 소문자 레이어명
            tilingScheme: new Cesium.GeographicTilingScheme(),
            parameters: {
                key: VWORLD_KEY,                  // sample.js 내에 선언된 전역 오픈 API 키 동기화
                domain: "127.0.0.1",              // CORS 방지용 브라우저 호스트 정착
                service: "WMS",
                version: "1.3.0",
                request: "GetMap",
                styles: "lt_c_spbd_nl",           // 성공 명세서 전용 스타일 지정
                crs: "EPSG:4326",                 // CRS:84 타일 조각 파싱 에러 강제 우회
                format: "image/png",              // 배경 투명 유지를 위해 png 포맷 필수
                transparent: "true"
            }
        });

        // 🚨 [핵심 오버라이드] 이미지 요청 주소에 브이월드 공식 보안 우회 프록시 게이트웨이를 강제로 주입합니다.
        disasterWmsProvider.requestImage = function (x, y, level, request) {
            const rect = this.tilingScheme.tileXYToNativeRectangle(x, y, level);

            // 1. 순수 WMS 데이터 타일 주소 빌드
            const baseUrl = "https://api.vworld.kr/req/wms";
            const queryParams = [
                `key=${VWORLD_KEY}`,
                `domain=127.0.0.1`,
                `service=WMS`,
                `version=1.3.0`,
                `request=GetMap`,
                `styles=lt_c_spbd_nl`,
                `crs=EPSG:4326`,
                `format=image/png`,
                `transparent=true`,
                `layers=lt_c_spbd`,
                `bbox=${rect.west},${rect.south},${rect.east},${rect.north}`,
                `width=256`,
                `height=256`
            ];

            const rawTileUrl = baseUrl + "?" + queryParams.join("&");

            // 2. 🎯 [보안 영구 해제 치트키] 브이월드가 자체 제공하는 자원 프록시 서버(proxy.do) 주소 뒤에 
            // 타겟 쿼리를 URL 인코딩하여 붙여 보냅니다. 이렇게 하면 브라우저가 동일 도메인 자원으로 인식합니다!
            const proxyGateUrl = "https://vworld.kr" + encodeURIComponent(rawTileUrl);

            // 3. 브이월드 서버를 직접 찌르지 않고 프록시 터널링을 경유하므로, crossOrigin 가드가 무조건 Pass 됩니다.
            return new Promise((resolve) => {
                const img = new Image();
                img.crossOrigin = "anonymous"; // 프록시 서버가 정상 헤더를 주므로 anonymous 설정 필수 가동

                img.onload = function () {
                    resolve(img); // 캔버스 세탁 과정 없이 즉시 깨끗한 상태로 3D 도면에 다이렉트 헌납
                };
                img.onerror = function () {
                    // 통신 누락 구역은 투명 버퍼로 안전 스킵하여 렌더러 보호
                    const empty = document.createElement("canvas");
                    empty.width = 256; empty.height = 256;
                    resolve(empty);
                };

                img.src = proxyGateUrl;
            });
        };

        // 가동기 내장 엔진의 XHR 강제 분기 차단 조치 추가
        disasterWmsProvider._tileDiscardPolicy = undefined;

        // 세슘 레이어 컬렉션에 추가하고 전역 변수에 인스턴스 바인딩 완수
        window.vworldDisasterLayerInstance = ws3d.viewer.imageryLayers.addImageryProvider(disasterWmsProvider);
        console.log("🟢 [CORS/Tainted 완벽 무력화] 브이월드 공식 프록시 게이트웨이를 경유하여 lt_c_spbd 레이어가 전체 화면에 도포됩니다.");
    } catch (e) {
        console.error("❌ 레이어 주입 중 크래시 발생:", e);
    }
};

        isShow = elementOrChecked.checked !== undefined ? elementOrChecked.checked : $(elementOrChecked).is(":checked");

window.drawFireZoneOnGround = function (lon, lat, isClickAction = false) {
    if (!ws3d || !ws3d.viewer) return;

    const liveLng = parseFloat(parseFloat(lon).toFixed(6));
    const liveLat = parseFloat(parseFloat(lat).toFixed(6));
    const wktPoint = "POINT(" + liveLng + " " + liveLat + ")";
    const timestamp = new Date().getTime();
    const viewer = ws3d.viewer;

    // 🚨 [목적지 설정 단추 트리거 실행부] 150m 절대 고도 스케일 원기둥 공역 인젝션 개시
    $.ajax({
        type: "get",
        url: "https://api.vworld.kr/req/data",
        data: { service: "data", request: "GetFeature", data: "LT_C_SPBD", key: VWORLD_KEY, domain: "127.0.0.1", geomFilter: wktPoint, buffer: "25", geometry: "true", crs: "EPSG:4326" },
        dataType: 'jsonp',
        jsonpCallback: "parseDirectDisaster_" + timestamp,
        success: function (data) {
            let bldName = "지정 관제 공역 부지";
            let rdsAddr = "지상 수평 좌표 공역 구역";
            let rawHeight = 0;

            if (data && data.response && data.response.status === "OK" && data.response.result?.featureCollection?.features) {
                let features = data.response.result.featureCollection.features;
                let featureObj = Array.isArray(features) ? features : features;
                if (featureObj && featureObj.properties) {
                    bldName = featureObj.properties.buld_nm || bldName;
                    rdsAddr = featureObj.properties.rds_rn_addr || rdsAddr;
                    rawHeight = featureObj.properties.buld_hg ? parseFloat(featureObj.properties.buld_hg) : 0;
                }
            }

            // 🚨 [요청 명세 정확 매핑] 목적지 설정 임무공역 고도는 '150m' 고정으로 하늘 높이 쭉 도출!
            const cylinderLength = 150.0;
            const centerAltitude = cylinderLength / 2.0;

            const buildingId = "DEST_" + Math.floor(liveLng * 100000);
            const targetEntityId = "vworld_fixed_disaster_zone_" + buildingId;
            const targetHtmlId = "vworld_html_popup_" + buildingId;

            // 목적지 토글 오프 해제 처리
            const existingFixedZone = viewer.entities.getById(targetEntityId);
            if (existingFixedZone) {
                viewer.entities.remove(existingFixedZone);
                const existPopup = document.getElementById(targetHtmlId);
                if (existPopup) { existPopup.parentNode.removeChild(existPopup); }
                window.gcsMissionNodes.destination = null;
                if (typeof window.refreshGcsMissionListUI === "function") window.refreshGcsMissionListUI();
                return;
            }

            // 전역 미션 시퀀스 목적지 노드 버퍼 동기화 적재
            window.gcsMissionNodes.destination = { lon: liveLng, lat: liveLat, alt: cylinderLength, name: bldName };

            const positionAnchor = Cesium.Cartesian3.fromDegrees(liveLng, liveLat, centerAltitude);
            const labelText = `[🎯 드론 목적지]\n위치: ${bldName}\n공역고도: ALT ${cylinderLength.toFixed(1)}m`;

            // 150m 원기둥 엔티티 렌더링 사양 조립
            const disasterEntityOptions = {
                id: targetEntityId,
                position: positionAnchor,
                cylinder: {
                    length: cylinderLength,
                    topRadius: 40.0, bottomRadius: 40.0,
                    material: Cesium.Color.RED.withAlpha(0.42), // 선명한 3D 반투명 빨간색 원기둥 투영 완수
                    outline: true, outlineColor: Cesium.Color.RED, outlineWidth: 3.0
                }
            };

            const addedEntity = viewer.entities.add(disasterEntityOptions);
            window.vworldFixedDisasterZonesMap[buildingId] = addedEntity;

            // 🚨 [요청 기능 완료 - 클릭 시 화면 텔레포트 flyTo 브릿지가 결합된 [X] 미니창]
            const existPopup = document.getElementById(targetHtmlId);
            if (existPopup) { existPopup.parentNode.removeChild(existPopup); }

            let mainMapDiv = document.getElementById("vmap") || document.getElementById("map") || viewer.scene.canvas.parentNode;
            const newPopupDiv = document.createElement("div");
            newPopupDiv.id = targetHtmlId;
            newPopupDiv.className = "gcs-custom-html-popup";
            newPopupDiv.setAttribute("data-lon", liveLng);
            newPopupDiv.setAttribute("data-lat", liveLat);

*/
window.gcsRegisterPostRenderListener = function (viewer) {
    if (!window.isDisasterRenderListenerRegistered) {
        window.isDisasterRenderListenerRegistered = true;
        viewer.scene.postRender.addEventListener(function () {
            document.querySelectorAll(".gcs-custom-html-popup").forEach(function (el) {
                const pLon = parseFloat(el.getAttribute("data-lon"));
                const pLat = parseFloat(el.getAttribute("data-lat"));
                const pAlt = parseFloat(el.getAttribute("data-alt")); const cartesianPos = Cesium.Cartesian3.fromDegrees(pLon, pLat, pAlt + 5.0);
                const screenPos = Cesium.SceneTransforms.wgs84ToWindowCoordinates(viewer.scene, cartesianPos); if (Cesium.defined(screenPos)) {
                    el.style.left = (screenPos.x - 25) + "px";
                    el.style.top = (screenPos.y - el.offsetHeight - 35) + "px";
                    el.style.display = "block";
                } else { el.style.display = "none"; }

// 전역 스코프에서 세슘 레이어 인스턴스를 추적하기 위한 가드 변수
window.vworldDisasterLayerInstance = null;

/**
 * 🎨 [기능 1] '소방방재 건물 레이어' 전체 화면에 타일 오버레이 자동 인젝션
 */
window.initDisasterWMSLayer = function () {
    if (typeof ws3d === "undefined" || !ws3d || !ws3d.viewer) {
        console.warn("⚠️ 세슘 엔진이 로드되지 않아 레이어를 주입할 수 없습니다.");
        return;
    }

    try {
        const layers = ws3d.viewer.imageryLayers;

        // 중복 방지 가드: 기존에 등록된 레이어가 있다면 안전하게 파괴 청소
        if (window.vworldDisasterLayerInstance) {
            layers.remove(window.vworldDisasterLayerInstance);
            window.vworldDisasterLayerInstance = null;
        }

        // 브이월드 표준 WMS 이미지 프로바이더 기동
        const disasterWmsProvider = new Cesium.WebMapServiceImageryProvider({
            url: "https://api.vworld.kr/req/wms",
            layers: "lt_c_spbd",
            tilingScheme: new Cesium.GeographicTilingScheme(),
            parameters: {
                key: VWORLD_KEY,
                domain: "127.0.0.1",
                service: "WMS",
                version: "1.3.0",
                request: "GetMap",
                styles: "lt_c_spbd_nl",
                crs: "EPSG:4326",
                format: "image/png",
                transparent: "true"
            }
        });

        // 🚨 [CORS 및 캔버스 오염 원천 박멸 인터셉터]
        disasterWmsProvider.requestImage = function (x, y, level, request) {
            const rect = this.tilingScheme.tileXYToNativeRectangle(x, y, level);
            const baseUrl = "https://api.vworld.kr/req/wms";
            const queryParams = [
                `key=${VWORLD_KEY}`, `domain=127.0.0.1`, `service=WMS`, `version=1.3.0`,
                `request=GetMap`, `styles=lt_c_spbd_nl`, `crs=EPSG:4326`, `format=image/png`,
                `transparent=true`, `layers=lt_c_spbd`,
                `bbox=${rect.west},${rect.south},${rect.east},${rect.north}`,
                `width=256`, `height=256`
            ];

            const rawTileUrl = baseUrl + "?" + queryParams.join("&");
            const proxyGateUrl = "https://map.vworld.kr/proxy.do?url=" + encodeURIComponent(rawTileUrl);

            return new Promise((resolve) => {
                const img = new Image();
                img.crossOrigin = "anonymous";

                img.onload = function () {
                    const canvas = document.createElement("canvas");
                    canvas.width = 256; canvas.height = 256;
                    canvas.getContext("2d").drawImage(img, 0, 0);
                    const cleanImg = new Image();
                    cleanImg.onload = function () { resolve(cleanImg); };
                    cleanImg.src = canvas.toDataURL("image/png");
                };
                img.onerror = function () {
                    const empty = document.createElement("canvas");
                    empty.width = 256; empty.height = 256;
                    resolve(empty);
                };
                img.src = proxyGateUrl;
            });
        };

        disasterWmsProvider._tileDiscardPolicy = undefined;

        // 🚨 [핵심 교정] addImageryProvider가 반환하는 실제 'ImageryLayer' 껍데기 인스턴스를 전역에 보관합니다.
        window.vworldDisasterLayerInstance = layers.addImageryProvider(disasterWmsProvider);
        console.log("🟢 [GCS 시각화] 파트 1: 소방방재 오버레이 레이어 적재 완료");
    } catch (e) {
        console.error("❌ 레이어 주입 중 크래시 발생:", e);
    }
};

 */
window.drawFireZoneOnGround = function (lon, lat, isClickAction = false) {
    if (!ws3d || !ws3d.viewer) return;

    const liveLng = parseFloat(parseFloat(lon).toFixed(6));
    const liveLat = parseFloat(parseFloat(lat).toFixed(6));
    const wktPoint = "POINT(" + liveLng + " " + liveLat + ")";
    const timestamp = new Date().getTime();
    const viewer = ws3d.viewer;

    // 단순 호버용 임시 띠 선제 청소
    const oldHoverPolyline = viewer.entities.getById("vworld_disaster_3d_polyline");
    if (oldHoverPolyline) { viewer.entities.remove(oldHoverPolyline); }

    $.ajax({
        type: "get",
        url: "https://api.vworld.kr/req/data",
        data: { service: "data", request: "GetFeature", data: "LT_C_SPBD", key: VWORLD_KEY, domain: "127.0.0.1", geomFilter: wktPoint, buffer: "25", geometry: "true", crs: "EPSG:4326" },
        dataType: 'jsonp',
        jsonpCallback: "parseDirectDisaster_" + timestamp,
        success: function (data) {
            let bldName = "GCS 지정 관제 건축물";
            let rdsAddr = "지상 수평 좌표 공역 구역";
            let rawHeight = 0;

            if (data && data.response && data.response.status === "OK" && data.response.result?.featureCollection?.features) {
                let features = data.response.result.featureCollection.features;
                let featureObj = Array.isArray(features) ? features : features;
                if (featureObj && featureObj.properties) {
                    bldName = featureObj.properties.buld_nm || bldName;
                    rdsAddr = featureObj.properties.rds_rn_addr || rdsAddr;
                    rawHeight = featureObj.properties.buld_hg ? parseFloat(featureObj.properties.buld_hg) : 0;
                }
            }

            // 🚨 [요청 사양 반영 1] 목적지 임무공역 및 가이드 기둥 길이를 '150.0m' 절대 초고층 규격으로 통일 주입!
            const cylinderLength = 150.0;
            const centerAltitude = cylinderLength / 2.0;

            const buildingId = "DEST_" + Math.floor(liveLng * 100000);
            const targetEntityId = "vworld_fixed_disaster_zone_" + buildingId;
            const targetHtmlId = "vworld_html_popup_" + buildingId;

            // 목적지 토글 오프 해제 처리 스레드
            if (isClickAction) {
                const existingFixedZone = viewer.entities.getById(targetEntityId);
                if (existingFixedZone) {
                    viewer.entities.remove(existingFixedZone);
                    const existPopup = document.getElementById(targetHtmlId);
                    if (existPopup) { existPopup.parentNode.removeChild(existPopup); }
                    window.gcsMissionNodes.destination = null;

            material: displayColor.withAlpha(0.40), outline: true, outlineColor: displayColor, outlineWidth: 3.0

    });

    let mainMapDiv = document.getElementById("vmap") || document.getElementById("map") || viewer.scene.canvas.parentNode;

    newPopupDiv.style.cssText = `position: absolute; z-index: 9999; background: #1e272e; border: 2px solid \${baseColorHex}; border-radius: 6px; padding: 10px; color: #fff; font-family: 'Malgun Gothic'; width: 180px; box-shadow: 0 4px 15px rgba(0, 0, 0, 0.5); pointer-events: auto; cursor: pointer;`;

/**
• 🚨 [요청 기능 핵심 교정 - 미니창 이름표 기준 완벽 정중앙(Center) 시선 카메라 무결점 텔레포트 오버라이드]
*/
window.gcsFlyToTargetCoordinates = function (lon, lat, alt) {
    if (!ws3d || !ws3d.viewer) return;
    console.log("✈️[GCS 센터 트래킹] 이름표 정중앙 매핑 조준 비행 점프 개시: 경도 ${ lon }, 위도 ${ lat }"); const targetLng = parseFloat(lon);
    const targetLat = parseFloat(lat);
    const targetAlt = parseFloat(alt);// 🎯 [수학적 센터 보정 치트키]
    // 기존에 카메라 각도(Pitch)를 비스듬히 눕히면 초점이 땅바닥을 찔러 글자가 모니터 하단 구석탱이로 밀려 내려갔던 것입니다.
    // 카메라가 바라보는 경사 각도(-45도 조감각)에 비례하여, 카메라의 위도(Lat) 물리 좌표를 남쪽 방향으로
    // 약 '0.0016도' 만큼 미세 오프셋 하강 이동시켜 렌더링함으로써 이름표 미니창이 정확히 브라우저 모니터 화면의
    // 정중앙 십자선 'Center 정중앙 복판'에 떡하니 배치 안착하도록 정밀 설계했습니다!
    const centerCorrectedLat = targetLat - 0.00165; ws3d.viewer.camera.flyTo({
        destination: Cesium.Cartesian3.fromDegrees(targetLng, centerCorrectedLat, targetAlt + 120.0), // 120m 상공 조감 고도 안착
        orientation: {
            heading: Cesium.Math.toRadians(0.0), // 정북 방향 유지
            pitch: Cesium.Math.toRadians(-45.0), // 가장 부드럽고 가시성이 뛰어난 -45도 정밀 조감 각도 셋팅
            roll: 0.0
        },
        duration: 1.0 // 1.0초 만에 시원하고 강력하게 광속 이동 완수
    });
};
/**
• 실시간 화면 스크린 모션 포스트 렌더 추적 엔진 통합 등록 모듈
*/
window.gcsRegisterPostRenderListener = function (viewer) {
    if (!window.isDisasterRenderListenerRegistered) {
        window.isDisasterRenderListenerRegistered = true;
        viewer.scene.postRender.addEventListener(function () {
            document.querySelectorAll(".gcs-custom-html-popup").forEach(function (el) {
                const pLon = parseFloat(el.getAttribute("data-lon"));
                const pLat = parseFloat(el.getAttribute("data-lat"));
                const pAlt = parseFloat(el.getAttribute("data-alt"));// 이름표가 기둥 머리맡 최상단에 찰떡처럼 붙어있도록 보정 마진 확보
                const cartesianPos = Cesium.Cartesian3.fromDegrees(pLon, pLat, pAlt + 4.0);
                const screenPos = Cesium.SceneTransforms.wgs84ToWindowCoordinates(viewer.scene, cartesianPos); if (Cesium.defined(screenPos)) {
                    el.style.left = (screenPos.x - 25) + "px";
                    el.style.top = (screenPos.y - el.offsetHeight - 35) + "px";
                    el.style.display = "block";
                } else { el.style.display = "none"; }
            });
        });
    }
};
/**
• 이름표 [X] 단추 클릭 시 메모리 풀 및 테이블 시퀀스 리스트에서 동시 완전 소멸
*/
window.removeDisasterZoneDirectly = function (uniqueKey) {
    if (!ws3d || !ws3d.viewer) return;
    const viewer = ws3d.viewer;
    const targetEntityId = "vworld_fixed_disaster_zone_" + uniqueKey;
    const targetHtmlId = "vworld_html_popup_" + uniqueKey; const fixedZoneEntity = viewer.entities.getById(targetEntityId);
    if (fixedZoneEntity) { viewer.entities.remove(fixedZoneEntity); } const popupEl = document.getElementById(targetHtmlId);
    if (popupEl) { popupEl.parentNode.removeChild(popupEl); } if (uniqueKey.startsWith("HOME")) {
        window.gcsMissionNodes.home = null;
    } else if (uniqueKey.startsWith("DEST")) {
        window.gcsMissionNodes.destination = null;
        window.vworldLastDisasterBuildingId = null;
    } else if (uniqueKey.startsWith("WAYPOINT")) {
        const wpLng = parseInt(uniqueKey.split("_")[1]) / 100000;
        window.gcsMissionNodes.waypoints = window.gcsMissionNodes.waypoints.filter(wp => {
            return Math.floor(wp.lon * 100000) !== Math.floor(wpLng * 100000);
        });
    } if (window.vworldFixedDisasterZonesMap) { delete window.vworldFixedDisasterZonesMap[uniqueKey]; }
    if (typeof window.refreshGcsMissionListUI === "function") window.refreshGcsMissionListUI();
    console.log("🧹[[X] 노드 삭제] GCS 오토파일럿 풀에서 ${ uniqueKey } 인프라 영구 소멸 완료.");
};

    /**
     * 🚨 [요청 기능 핵심 - 화면 초고속 텔레포트 스레드]
     * 이름표(미니창) 패널을 마우스로 클릭하는 즉시 세슘 카메라 광선을 구동해 해당 좌표 상공으로 순간 이동 비행 시킵니다.
     */
    window.gcsFlyToTargetCoordinates = function(lon, lat, alt) {
        if (!ws3d || !ws3d.viewer) return;
        console.log(`✈️ [GCS 플라이트 컨트롤] 해당 이름표 목적지 공간으로 즉시 카메라 뷰포트 이동: 경도 ${lon}, 위도 ${lat}`);
        
        ws3d.viewer.camera.flyTo({
            destination: Cesium.Cartesian3.fromDegrees(parseFloat(lon), parseFloat(lat), parseFloat(alt) + 180.0), // 보기 좋게 상공 180m 지점으로 카메라 안착
            orientation: {
                heading: Cesium.Math.toRadians(0.0),
                pitch: Cesium.Math.toRadians(-55.0), // 비스듬히 아래를 굽어보도록 시선 각도 정밀 동기화
                roll: 0.0
            },
            duration: 1.2 // 1.2초 만에 부드럽고 신속하게 이동 완수
        });
    };

    // 화면 좌클릭 시 우클릭 창 숨기기 가드
    document.addEventListener("click", function() {
        const menu = document.getElementById("gcs_context_menu");
        if (menu) menu.style.display = "none";
    });
